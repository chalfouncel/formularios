async function gerarPDF() { const { jsPDF } = window.jspdf; const pdf = new jsPDF('p', 'mm', 'a4'); const logoUrl = 'https://raw.githubusercontent.com/chalfouncel/formularios/main/Loga%20marca%20nova.png'; try { const logoImg = await carregarImagem(logoUrl); // Cabeçalho na primeira página pdf.setFillColor(26, 26, 26); pdf.rect(0, 0, 210, 45, 'F'); pdf.addImage(logoImg, 'PNG', 85, 5, 40, 20); pdf.setTextColor(212, 175, 55); pdf.setFontSize(18); pdf.setFont(undefined, 'bold'); pdf.text('M&IC CORRETORES DE IMÓVEIS', 105, 32, { align: 'center' }); pdf.setTextColor(200, 200, 200); pdf.setFontSize(10); pdf.setFont(undefined, 'normal'); pdf.text('Relatório de Documentos — ' + perfilAtual.nome, 105, 39, { align: 'center' }); pdf.setTextColor(120, 120, 120); pdf.setFontSize(8); pdf.text(new Date().toLocaleString('pt-BR'), 105, 43, { align: 'center' }); let y = 55; // Página de dados do proprietário if (dadosProprietario) { pdf.addPage(); y = 20; pdf.setFillColor(26, 26, 26); pdf.rect(0, y - 5, 210, 12, 'F'); pdf.setTextColor(212, 175, 55); pdf.setFontSize(14); pdf.setFont(undefined, 'bold'); pdf.text('DADOS DO PROPRIETÁRIO', 105, y + 3, { align: 'center' }); y += 15; pdf.setTextColor(26, 26, 26); pdf.setFontSize(11); pdf.setFont(undefined, 'bold'); pdf.text('Informações pessoais', 14, y); y += 6; pdf.setFontSize(9); pdf.setFont(undefined, 'normal'); pdf.setTextColor(60, 60, 60); const dadosPessoais = [ ['Finalidade:', dadosProprietario.finalidade === 'locacao' ? 'Locação' : 'Venda'], ['Nome:', dadosProprietario.nome], ['Nacionalidade:', dadosProprietario.nacionalidade], ['Estado Civil:', dadosProprietario.estadoCivil], ['Profissão:', dadosProprietario.profissao], ['RG:', dadosProprietario.rg || '-'], ['CPF:', dadosProprietario.cpf], ['Endereço:', dadosProprietario.endereco], ['E-mail:', dadosProprietario.email], ['Telefone:', dadosProprietario.telefone], ['Endereço do imóvel:', dadosProprietario.enderecoImovel] ]; dadosPessoais.forEach(([label, value]) => { if (y > 275) { pdf.addPage(); y = 20; } pdf.setFont(undefined, 'bold'); pdf.text(label, 14, y); pdf.setFont(undefined, 'normal'); const lines = pdf.splitTextToSize(value, 120); pdf.text(lines, 60, y); y += 5 * lines.length + 1; }); if (dadosProprietario.finalidade === 'venda') { y += 3; if (y > 275) { pdf.addPage(); y = 20; } pdf.setTextColor(26, 26, 26); pdf.setFontSize(11); pdf.setFont(undefined, 'bold'); pdf.text('Dados do imóvel (venda)', 14, y); y += 6; pdf.setFontSize(9); pdf.setFont(undefined, 'normal'); pdf.setTextColor(60, 60, 60); const dadosVenda = [ ['IPTU:', dadosProprietario.iptu], ['Matrícula RGI:', dadosProprietario.matricula], ['Cartório RGI:', dadosProprietario.rgi], ['Financiado:', dadosProprietario.financiado === 'sim' ? 'Sim' : 'Não'] ]; if (dadosProprietario.financiado === 'sim') { dadosVenda.push(['Banco:', dadosProprietario.bancoFinanciamento]); dadosVenda.push(['Saldo devedor:', dadosProprietario.saldoDevedor]); } dadosVenda.push(['FGTS:', dadosProprietario.fgts === 'sim' ? 'Sim' : 'Não']); dadosVenda.forEach(([label, value]) => { if (y > 275) { pdf.addPage(); y = 20; } pdf.setFont(undefined, 'bold'); pdf.text(label, 14, y); pdf.setFont(undefined, 'normal'); const lines = pdf.splitTextToSize(value, 120); pdf.text(lines, 60, y); y += 5 * lines.length + 1; }); } y += 3; if (y > 275) { pdf.addPage(); y = 20; } pdf.setTextColor(26, 26, 26); pdf.setFontSize(11); pdf.setFont(undefined, 'bold'); pdf.text('Dados bancários', 14, y); y += 6; pdf.setFontSize(9); pdf.setFont(undefined, 'normal'); pdf.setTextColor(60, 60, 60); const dadosBancarios = [ ['Banco:', dadosProprietario.banco], ['Agência:', dadosProprietario.agencia], ['Conta:', dadosProprietario.conta], ['Chave PIX:', dadosProprietario.chavePix] ]; dadosBancarios.forEach(([label, value]) => { if (y > 275) { pdf.addPage(); y = 20; } pdf.setFont(undefined, 'bold'); pdf.text(label, 14, y); pdf.setFont(undefined, 'normal'); pdf.text(value, 60, y); y += 6; }); } // Página de documentos if (docsCapturados.length > 0) { pdf.addPage(); y = 20; pdf.setFillColor(26, 26, 26); pdf.rect(0, y - 5, 210, 12, 'F'); pdf.setTextColor(212, 175, 55); pdf.setFontSize(14); pdf.setFont(undefined, 'bold'); pdf.text('DOCUMENTOS CAPTURADOS', 105, y + 3, { align: 'center' }); y += 15; for (let i = 0; i < docsCapturados.length; i++) { const doc = docsCapturados[i]; let imgW = 170, imgH = 0, temImagem = false, formato = 'JPEG'; if (doc.foto && doc.foto.length > 100) { try { const props = pdf.getImageProperties(doc.foto); if (doc.foto.indexOf('data:image/png') === 0) formato = 'PNG'; imgW = 170; imgH = (props.height * imgW) / props.width; if (imgH > 180) { imgH = 180; imgW = (props.width * imgH) / props.height; } temImagem = true; } catch (e) { temImagem = false; } } const alturaBloco = 14 + (temImagem ? imgH + 10 : 12); // Verifica se cabe na página atual if (y + alturaBloco > 285) { pdf.addPage(); y = 20; } pdf.setFillColor(212, 175, 55); pdf.rect(12, y - 4, 186, 8, 'F'); pdf.setTextColor(26, 26, 26); pdf.setFontSize(10); pdf.setFont(undefined, 'bold'); pdf.text((i + 1) + '. ' + doc.nome, 16, y + 2); y += 14; if (temImagem) { const x = (210 - imgW) / 2; pdf.addImage(doc.foto, formato, x, y, imgW, imgH); y += imgH + 10; } else { pdf.setTextColor(100, 100, 100); pdf.setFontSize(9); pdf.setFont(undefined, 'normal'); pdf.text('[' + doc.nomeArquivo + ' — arquivo anexado]', 105, y, { align: 'center' }); y += 12; } } } // Remove a última página se estiver vazia const totalPages = pdf.internal.getNumberOfPages(); if (totalPages > 1) { const lastPage = pdf.internal.pages[totalPages]; if (lastPage && lastPage.length <= 2) { pdf.deletePage(totalPages); } } pdf.save('M&IC_' + perfilAtual.nome + '_documentos.pdf'); } catch (err) { console.error('Erro ao gerar PDF:', err); alert('Erro ao gerar PDF. Detalhe: ' + err.message); } }
