async function gerarPDF() { const { jsPDF } = window.jspdf; const pdf = new jsPDF('p', 'mm', 'a4'); const logoUrl = 'https://raw.githubusercontent.com/chalfouncel/formularios/main/Loga%20marca%20nova.png'; try { const logoImg = await carregarImagem(logoUrl); // Cabeçalho pdf.setFillColor(26, 26, 26); pdf.rect(0, 0, 210, 45, 'F'); pdf.addImage(logoImg, 'PNG', 85, 5, 40, 20); pdf.setTextColor(212, 175, 55); pdf.setFontSize(18); pdf.setFont(undefined, 'bold'); pdf.text('M&IC CORRETORES DE IMÓVEIS', 105, 32, { align: 'center' }); pdf.setTextColor(200, 200, 200); pdf.setFontSize(10); pdf.setFont(undefined, 'normal'); pdf.text('Relatório de Documentos — ' + perfilAtual.nome, 105, 39, { align: 'center' }); pdf.setTextColor(120, 120, 120); pdf.setFontSize(8); pdf.text(new Date().toLocaleString('pt-BR'), 105, 43, { align: 'center' }); let y = 55; for (let i = 0; i < docsCapturados.length; i++) { const doc = docsCapturados[i]; // ---- PRÉ-CALCULAR altura da imagem ---- let imgW = 170, imgH = 0, temImagem = false, formato = 'JPEG'; if (doc.foto && doc.foto.length > 100) { try { const props = pdf.getImageProperties(doc.foto); if (doc.foto.indexOf('data:image/png') === 0) formato = 'PNG'; imgW = 170; imgH = (props.height * imgW) / props.width; // Reduzir altura máxima para caber em uma folha if (imgH > 180) { imgH = 180; imgW = (props.width * imgH) / props.height; } temImagem = true; } catch (e) { temImagem = false; } } // ---- Calcular altura TOTAL do bloco (título + imagem) ---- const alturaBloco = 14 + (temImagem ? imgH + 10 : 12); // ---- Se NÃO couber, criar nova página ANTES do título ---- if (y + alturaBloco > 285) { pdf.addPage(); y = 20; } // ---- Desenhar título (nunca separado da imagem) ---- pdf.setFillColor(212, 175, 55); pdf.rect(12, y - 4, 186, 8, 'F'); pdf.setTextColor(26, 26, 26); pdf.setFontSize(10); pdf.setFont(undefined, 'bold'); pdf.text((i + 1) + '. ' + doc.nome, 16, y + 2); y += 14; // ---- Desenhar imagem (sempre junto com o título) ---- if (temImagem) { const x = (210 - imgW) / 2; pdf.addImage(doc.foto, formato, x, y, imgW, imgH); y += imgH + 10; } else { pdf.setTextColor(100, 100, 100); pdf.setFontSize(9); pdf.setFont(undefined, 'normal'); pdf.text('[' + doc.nomeArquivo + ' — arquivo anexado]', 105, y, { align: 'center' }); y += 12; } } pdf.save('M&IC_' + perfilAtual.nome + '_documentos.pdf'); } catch (err) { console.error('Erro ao gerar PDF:', err); alert('Erro ao gerar PDF. Detalhe: ' + err.message); } }
